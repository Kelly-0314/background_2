def check_main_ahead_of_original_strict(student_owner: str) -> (bool, str):
    """
    Rule:
      - If repo is template-generated → immediate FAIL with a clear message.
      - If repo is not a fork of the upstream → FAIL (must fork the official repo).
      - If repo is a fork → compare upstream:main...student:main and require ahead_by >= 1.
    """
    # Fetch repo info
    r = _gh_get(f"https://api.github.com/repos/{student_owner}/{FORK_REPO}")
    _require_ok(r, "fetching repo info")
    repo_info = r.json()

    # Template-generated repos are not allowed for this exercise
    if repo_info.get("template_repository"):
        tmpl = repo_info["template_repository"]
        tmpl_owner = tmpl["owner"]["login"]
        tmpl_name = tmpl["name"]
        return (
            False,
            f"❌ '{student_owner}/{FORK_REPO}' was generated from template "
            f"'{tmpl_owner}/{tmpl_name}', not forked. This exercise requires you to "
            f"fork '{UPSTREAM_OWNER}/{UPSTREAM_REPO}'. Please recreate your work from a fork."
        )

    # Must be a fork
    if not repo_info.get("fork"):
        return (
            False,
            f"❌ '{student_owner}/{FORK_REPO}' is not a fork of the official repository. "
            f"Please fork '{UPSTREAM_OWNER}/{UPSTREAM_REPO}'."
        )

    # Verify parent is the expected upstream (or at least compare against the parent)
    parent = repo_info.get("parent")
    if not parent:
        return (
            False,
            f"❌ Could not identify the parent repository for '{student_owner}/{FORK_REPO}'."
        )
    base_owner = parent["owner"]["login"]
    base_repo  = parent["name"]

    # Compare upstream main vs student main
    compare_url = (
        f"https://api.github.com/repos/{base_owner}/{base_repo}/compare/"
        f"{MAIN_BRANCH}...{student_owner}:{MAIN_BRANCH}"
    )
    r = _gh_get(compare_url)
    if r.status_code == 404:
        return (
            False,
            f"❌ Could not compare '{base_owner}/{base_repo}:{MAIN_BRANCH}' to "
            f"'{student_owner}/{FORK_REPO}:{MAIN_BRANCH}'. Double-check branch names."
        )
    _require_ok(r, "comparing upstream and fork")
    data = r.json()

    ahead_by = data.get("ahead_by", 0)
    if ahead_by >= 1:
        return True, (
            f"✅ Fork '{student_owner}/{FORK_REPO}:{MAIN_BRANCH}' is {ahead_by} "
            f"commit(s) ahead of '{base_owner}/{base_repo}:{MAIN_BRANCH}'."
        )
    else:
        return (
            False,
            f"⏳ Fork '{student_owner}/{FORK_REPO}:{MAIN_BRANCH}' is not ahead of upstream "
            f"(ahead_by={ahead_by}). Push at least one new commit."
        )

def check_ex1_merged_into_main(fork_owner: str) -> (bool, str):
    """
    Check that ex1 was merged into main in the student's repo.
    Strategy:
      - Compare base=main...head=ex1 on the fork. If ahead_by == 0 → ex1 fully contained in main.
      - If ex1 branch is missing (404), look for a merged PR from ex1 → main.
    """
    compare_url = (
        f"https://api.github.com/repos/{fork_owner}/{FORK_REPO}/compare/"
        f"{MAIN_BRANCH}...{EX1_BRANCH}"
    )
    r = _gh_get(compare_url)
    if r.status_code == 404:
        # ex1 might have been deleted post-merge; check for a merged PR
        pulls_url = f"https://api.github.com/repos/{fork_owner}/{FORK_REPO}/pulls"
        r2 = _gh_get(
            pulls_url,
            params={
                "state": "closed",
                "base": MAIN_BRANCH,
                "head": f"{fork_owner}:{EX1_BRANCH}",
                "per_page": 100,
            },
        )
        _require_ok(r2, "listing closed PRs for ex1 → main")
        closed_prs = r2.json()
        merged = any(pr.get("merged_at") for pr in closed_prs)
        if merged:
            return True, (
                f"✅ 'ex1' appears merged into '{MAIN_BRANCH}' "
                f"(branch missing but a PR from ex1 → {MAIN_BRANCH} is merged)."
            )
        else:
            return False, (
                f"⏳ Could not compare and found no merged PR from '{EX1_BRANCH}' to '{MAIN_BRANCH}'."
            )

    _require_ok(r, "comparing main...ex1 on the fork")
    data = r.json()
    ahead_by = data.get("ahead_by", None)
    if ahead_by is None:
        return False, "⏳ Unexpected compare payload (no 'ahead_by')."
    if ahead_by == 0:
        return True, f"✅ Branch '{EX1_BRANCH}' is fully merged into '{MAIN_BRANCH}'."
    else:
        return False, (
            f"⏳ Branch '{EX1_BRANCH}' still has {ahead_by} commit(s) not in '{MAIN_BRANCH}'."
        )

def check_notebook_has_no_conflicts(fork_owner: str) -> (bool, str):
    """
    Ensure the notebook can be fetched from main, parses as JSON,
    and has no conflict markers anywhere (including within code cells).
    """
    raw_url = (
        f"https://raw.githubusercontent.com/{fork_owner}/{FORK_REPO}/"
        f"{MAIN_BRANCH}/{NOTEBOOK_PATH}"
    )
    r = requests.get(raw_url, timeout=30)
    if r.status_code != 200:
        return False, (
            f"❌ Could not access '{NOTEBOOK_PATH}' in '{fork_owner}/{FORK_REPO}' "
            f"(HTTP {r.status_code})."
        )
    text = r.text

    # Quick scan for conflict markers in raw text
    if any(marker in text for marker in ("<<<<<<<", "=======", ">>>>>>>" )):
        return False, (
            f"⏳ Merge-conflict markers found in '{NOTEBOOK_PATH}'. Please resolve them."
        )

    # Validate JSON and re-check within cells
    try:
        nb = json.loads(text)
    except Exception as e:
        return False, (
            f"⏳ '{NOTEBOOK_PATH}' is not valid JSON (possibly unresolved conflicts). "
            f"Error: {e}"
        )

    conflict_re = re.compile(r"(<<<<<<<|=======|>>>>>>>)")
    for cell in nb.get("cells", []):
        src = "".join(cell.get("source", [])) if isinstance(cell.get("source"), list) else cell.get("source", "")
        if isinstance(src, str) and conflict_re.search(src):
            return False, (
                f"⏳ Conflict markers remain inside cells of '{NOTEBOOK_PATH}'."
            )

    return True, f"✅ No conflict markers found in '{NOTEBOOK_PATH}'."

def run_all_checks(github_username: str):
    ok1, msg1 = check_main_ahead_of_original_strict(github_username)
    ok2, msg2 = check_ex1_merged_into_main(github_username)
    ok3, msg3 = check_notebook_has_no_conflicts(github_username)

    all_ok = ok1 and ok2 and ok3
    return all_ok, "\n".join([msg1, msg2, msg3])
